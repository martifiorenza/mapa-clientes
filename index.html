onEachFeature: function (f, layer) { const p = f.properties; // Buscamos los valores sin importar Mayus/Minus const nombre = limpiar(obtenerValor(p, "cliente"), "Empresa sin nombre"); const localidad = limpiar(obtenerValor(p, "localidad"), ""); const provincia = limpiar(obtenerValor(p, "provincia"), ""); const crm = limpiar(obtenerValor(p, "crm"), "NO"); // Detectamos si es CAPIA buscando el campo "Cuit" o "RELEVADO" const esCapia = obtenerValor(p, "Cuit") !== null || obtenerValor(p, "RELEVADO") !== null; let contenidoPopup = `
${nombre}
📍 Ubicación:
${localidad}${localidad && provincia ? ', ' : ''}${provincia}
`; if (esCapia) { // --- DISEÑO PARA CAPIA --- const cuit = limpiar(obtenerValor(p, "Cuit"), "No registrado"); const relevado = limpiar(obtenerValor(p, "RELEVADO"), "NO"); contenidoPopup += `
🆔 CUIT: ${cuit}
✅ Relevado: ${relevado}
`; } else { // --- DISEÑO PARA BASE 1 (MAQUINARIA) --- const maquinas = limpiar(obtenerValor(p, "maquinas"), "Sin datos"); const cantidad = limpiar(obtenerValor(p, "cantidad"), "0"); contenidoPopup += `
🚜 Máquinas: ${maquinas}
🔢 Cantidad: ${cantidad}
`; } // El pie del popup con el CRM (común a ambos) contenidoPopup += `
📊 CRM: ${crm}
`; layer.bindPopup(contenidoPopup); }
